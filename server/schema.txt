create or replace function public.validate_subscription_unit()
returns trigger
language plpgsql
set search_path = ''
as $$
declare
  max_units integer;
begin
  select s.total_units into max_units
  from public.sites s
  where s.id = new.site_id;

  if max_units is null then
    raise exception 'Unknown solar site';
  end if;

  if new.unit_index < 0 or new.unit_index >= max_units then
    raise exception 'Unit index is outside the site capacity';
  end if;

  return new;
end;
$$;

create trigger validate_subscription_unit_before_write
before insert or update on public.subscription_units
for each row execute function public.validate_subscription_unit();

-- 10. Secure points and check-in functions
-- Do not grant these to anon. Authenticated users may execute only
-- functions that award fixed, server-defined points.
-- ----------------------------
create or replace function public.apply_points(
  p_user_id uuid,
  p_delta integer,
  p_activity public.activity_type,
  p_title text,
  p_idempotency_key text,
  p_metadata jsonb default '{}'::jsonb
)
returns uuid
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_id uuid;
  v_balance integer;
begin
  if p_delta = 0 then
    raise exception 'Points delta cannot be zero';
  end if;

  -- Lock profile row so concurrent redemptions cannot overspend.
  select points_balance into v_balance
  from public.profiles
  where id = p_user_id
  for update;

  if not found then
    raise exception 'Profile not found';
  end if;

  if v_balance + p_delta < 0 then
    raise exception 'Insufficient points';
  end if;

  insert into public.point_ledger
    (user_id, activity, title, points_delta, idempotency_key, metadata)
  values
    (p_user_id, p_activity, p_title, p_delta, p_idempotency_key, coalesce(p_metadata, '{}'::jsonb))
  on conflict (idempotency_key) do nothing
  returning id into v_id;

  if v_id is null then
    return null; -- already processed; caller should treat as duplicate
  end if;

  update public.profiles
  set points_balance = points_balance + p_delta,
      points_lifetime = points_lifetime + greatest(p_delta, 0)
  where id = p_user_id;

  return v_id;
end;
$$;

create or replace function public.check_in_today()
returns uuid
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_user uuid := auth.uid();
  v_inserted uuid;
begin
  if v_user is null then
    raise exception 'Authentication required';
  end if;

  insert into public.daily_checkins(user_id, checkin_date)
  values (v_user, (now() at time zone 'Asia/Kuala_Lumpur')::date)
  on conflict do nothing;

  if not found then
    raise exception 'Already checked in today';
  end if;

  v_inserted := public.apply_points(
    v_user, 10, 'checkin', 'Daily check-in',
    'checkin:' || v_user::text || ':' ||
      (now() at time zone 'Asia/Kuala_Lumpur')::date::text,
    jsonb_build_object('source', 'daily_checkin')
  );

  return v_inserted;
end;
$$;

create or replace function public.redeem_reward(p_reward_id text)
returns uuid
language plpgsql
security definer
set search_path = ''
as $$
declare
  v_user uuid := auth.uid();
  v_reward public.rewards%rowtype;
  v_redemption_id uuid;
begin
  if v_user is null then
    raise exception 'Authentication required';
  end if;

  select * into v_reward
  from public.rewards
  where id = p_reward_id and active = true
  for share;

  if not found then
    raise exception 'Reward unavailable';
  end if;

  -- apply_points locks the profile row and rejects overspending.
  perform public.apply_points(
    v_user, -v_reward.cost_points, 'redeem',
    'Redeemed reward: ' || v_reward.title,
    'redeem:' || gen_random_uuid()::text,
    jsonb_build_object('reward_id', v_reward.id)
  );

  insert into public.redemptions(user_id, reward_id, points_spent)
  values (v_user, v_reward.id, v_reward.cost_points)
  returning id into v_redemption_id;

  return v_redemption_id;
end;
$$;

-- 12. Portfolio summary view
-- Use transaction tables for actual amounts, not unit price estimates.
-- ----------------------------
create view public.v_my_portfolio_summary
with (security_invoker = true)
as
select
  p.id as user_id,
  coalesce(c.total_contributions, 0)::numeric(14,2) as completed_contributions_myr,
  coalesce(py.total_payouts, 0)::numeric(14,2) as completed_payouts_myr,
  coalesce(c.total_contributions, 0)::numeric(14,2) as contribution_basis_myr
from public.profiles p
left join (
  select user_id, sum(amount) as total_contributions
  from public.contributions
  where status = 'completed'
  group by user_id
) c on c.user_id = p.id
left join (
  select user_id, sum(amount) as total_payouts
  from public.payouts
  where status = 'completed'
  group by user_id
) py on py.user_id = p.id;

grant select on public.v_my_portfolio_summary to authenticated;
